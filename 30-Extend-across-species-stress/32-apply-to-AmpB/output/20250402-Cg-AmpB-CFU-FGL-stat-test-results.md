  select(-p_intermediate, -p_dead) %>% # remove unwanted variables
  pivot_wider(names_from = Method, values_from = p_live) %>% 
  dplyr::filter(!Treatment %in% c("0", "0.02")) # remove mock and 0.02 ng/mL

  Pearson = cor(tmp$CFU, tmp$FungaLight, method = "pearson"),
